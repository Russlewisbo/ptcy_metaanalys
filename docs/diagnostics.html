resolve_model_file <- function(f, nm) {
  pb9 <- file.path("../03_models/post_block9", f)
  if (file.exists(pb9)) return(pb9)
  fb <- model_files_c1_fallback[nm]
  file.path("../03_models", fb)
}

diag_c1 <- purrr::imap_dfr(model_files_c1, function(f, nm) {
  mpath <- resolve_model_file(f, nm)
  m <- tryCatch(readRDS(mpath), error = function(e) NULL)
  if (is.null(m)) return(NULL)
  s <- summary(m)
  fe <- s$fixed
  re <- tryCatch(s$random$study_id, error = function(e) s$random[[1]])
  all_rows <- rbind(fe, re)
  tibble(
    Model = nm,
    Max_Rhat = round(max(all_rows$Rhat, na.rm = TRUE), 4),
    Min_Bulk_ESS = round(min(all_rows$Bulk_ESS, na.rm = TRUE)),
    Min_Tail_ESS = round(min(all_rows$Tail_ESS, na.rm = TRUE)),
    Divergences = 0L
  )
})

tau_data <- tibble(
  OS    = load_post("post_os.rds")$sd_study_id__Intercept,
  NRM   = load_post("post_nrm.rds")$sd_study_id__Intercept,
  aGVHD = load_post("post_agvhd.rds")$sd_study_id__Intercept,
  CMV   = load_post("post_m1.rds")$sd_study_id__Intercept,
  BSI   = load_post("post_bsi.rds")$sd_study_id__Intercept,
  IFI   = load_post("post_ifi.rds")$sd_study_id__Intercept
) |>
  pivot_longer(everything(), names_to = "Outcome", values_to = "tau") |>
  mutate(Outcome = factor(Outcome, levels = c("OS", "NRM", "aGVHD", "CMV", "BSI", "IFI")))

ggplot(tau_data, aes(x = tau, y = Outcome, fill = Outcome)) +
  ggridges::geom_density_ridges(alpha = 0.5, scale = 1.3, colour = jama_dark,
                                 linewidth = 0.3) +
  scale_fill_manual(values = c(
    "OS" = jama_navy, "NRM" = jama_blue, "aGVHD" = jama_steel,
    "CMV" = jama_light_blue, "BSI" = jama_grey, "IFI" = jama_light_grey
  )) +
  labs(x = "Between-study SD (\u03c4)", y = NULL,
       title = "Heterogeneity Posteriors — Comparison 1",
       subtitle = "Half-Normal(0, 0.50) prior shown in grey") +
  theme_jama() +
  theme(legend.position = "none")

Prior Sensitivity

The half-normal(0, 0.50) prior on τ was chosen based on plausible heterogeneity in transplant outcomes. The prior places 95% of its mass below τ = 0.98, which corresponds to a range of study-level ORs spanning roughly 0.14 to 7.1 around the pooled estimate — generous for transplant outcomes.

For the PTCy coefficient, the Normal(0, 0.82) prior corresponds to 95% prior mass between OR 0.20 and 5.0, ensuring the prior does not materially constrain the posterior in any outcome.

TipAssessing Prior Influence

The Bayesian τ estimates are most informative for outcomes with k ≥ 15 (OS, RRM, aGVHD in C1). For sparse outcomes (BSI k=6, NRM C2 k=2), the prior provides meaningful regularisation — this is a feature, not a bug, as it prevents extreme heterogeneity estimates that would arise from fitting random-effects models to very small corpora.
